#    public name from the Elastic IP, which is not in TEAM_IPS, so a team-only staging host would answer 403
#    to the deployment's own health check;
#  - the grader opens both URLs.
# The names pricing-staging and pricing-production are network aliases set by ansible/templates/compose.yaml.j2
# on the existing gitea_web network. Port 8000 is never published on the host.

	reverse_proxy pricing-staging:8000
}

app.ssw590-g1.tech {
	reverse_proxy pricing-production:8000
